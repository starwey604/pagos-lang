cmake_minimum_required(VERSION 3.30)
project(PagosFileCheck LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(../../cmake/llvm_setup.cmake)
if(NOT EXISTS "${PAGOS_FILECHECK_SOURCE}")
    message(FATAL_ERROR "Supply the pinned upstream FileCheck.cpp with PAGOS_FILECHECK_SOURCE")
endif()
add_executable(FileCheck "${PAGOS_FILECHECK_SOURCE}")
pagos_use_llvm_headers(FileCheck)
target_compile_options(FileCheck PRIVATE ${PAGOS_LLVM_DEFINITIONS})
target_link_libraries(FileCheck PRIVATE LLVMFileCheck)
if(MSVC)
    target_compile_options(FileCheck PRIVATE /EHsc /utf-8 /bigobj)
endif()
